  <action>
    Create backend/rehab_robotics_bridge/opensim/n_sensor_calibration.py.

    Class CalibrationArtifactStore with no __init__ parameters (stateless utility class).

    CalibrationArtifactStore.compute_artifact_path(model_hash: str, applied_revision: int) -> Path:
      base = Path.home() / ".ros" / "rehab_robotics"
      hash8 = str(model_hash)[:8]
      filename = f"calibration_{hash8}_rev{applied_revision}.json"
      return base / filename

    CalibrationArtifactStore.save(artifact_path: Path, data: dict) -> None:
      artifact_path.parent.mkdir(parents=True, exist_ok=True)
      serialised = json.dumps(data, sort_keys=True, indent=2)
      tmp = artifact_path.with_suffix(".tmp")
      tmp.write_text(serialised, encoding="utf-8")
      tmp.replace(artifact_path)

    CalibrationArtifactStore.load(artifact_path: Path) -> dict | None:
      try:
        text = artifact_path.read_text(encoding="utf-8")
        data = json.loads(text)
        if not isinstance(data, dict): return None
        if data.get("schema_version") != "calib.v1": return None
        return data
      except Exception: return None

    CalibrationArtifactStore.is_valid(
        artifact: dict | None,
        current_model_hash: str,
        current_applied_revision: int,
        current_device_order: list[str],
    ) -> bool:
      if artifact is None: return False
      if artifact.get("model_hash") != current_model_hash: return False
      if int(artifact.get("applied_revision", -1)) != current_applied_revision: return False
      stored_devices = set(artifact.get("device_order", []))
      current_devices = set(current_device_order)
      return stored_devices == current_devices

    Export __all__ = ["CalibrationArtifactStore"].

    Tests use tmp_path fixture (pytest) or tempfile.TemporaryDirectory for isolation.

    Implement _on_calibration_capture_n(self, _request, response) -> response:
      Under self._input_lock:
        inputs = dict(self._mac_inputs)   # snapshot
      if not inputs:
        response.success = False
        response.message = json.dumps({"outcome": "no_mapping", "artifact_path": None})
        return response
      missing = [did for did, entry in inputs.items() if entry.last_xyzw is None]
      if missing:
        response.success = False
        response.message = json.dumps({"outcome": "missing_inputs", "artifact_path": None})
        return response
      # Build artifact
      device_order = sorted(inputs.keys())   # D-03 alphabetical sort
      frame_assignments = {did: {"segment": "", "frame": inputs[did].frame} for did in device_order}
      reference_pose = {}
      for did in device_order:
        xyzw = inputs[did].last_xyzw
        reference_pose[did] = {"qw": xyzw[3], "qx": xyzw[0], "qy": xyzw[1], "qz": xyzw[2]}
      artifact = {
        "schema_version": "calib.v1",
        "model_hash": self._n_model_hash,
        "applied_revision": self._n_mapping_revision,
        "device_order": device_order,
        "frame_assignments": frame_assignments,
        "solver_profile": "lower_body",
        "calibrated_at_iso8601": datetime.datetime.utcnow().isoformat() + "Z",
        "reference_pose": reference_pose,
      }
      artifact_path = self._n_calib_store.compute_artifact_path(
          self._n_model_hash, self._n_mapping_revision
      )
      self._n_calib_store.save(artifact_path, artifact)
      self._n_calib_artifact = artifact
      self._n_calib_state = "calibrated"
      self._publish_n_calibration_status()
      response.success = True
      response.message = json.dumps({"outcome": "captured", "artifact_path": str(artifact_path)})
      return response

    Implement _check_artifact_validity(self) -> None:
      if self._n_calib_artifact is None: return
      with self._input_lock:
        current_device_order = sorted(self._mac_inputs.keys())
      valid = self._n_calib_store.is_valid(
          self._n_calib_artifact,
          self._n_model_hash,
          self._n_mapping_revision,
          current_device_order,
      )
      if not valid:
        self._n_calib_artifact = None
        self._n_calib_state = "uncalibrated"
        self._publish_n_calibration_status()
        self.get_logger().info("N-sensor calibration invalidated by mapping change")

    Call _check_artifact_validity() at end of _on_mapping_current, after updating _mac_inputs.
    Also update _n_model_hash and _n_mapping_revision from the mapping JSON in _on_mapping_current.

      payload = {

      self._n_calibration_status_publisher.publish(message)

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-23-02-01 | Tampering | Artifact file on disk | accept | User-owned directory; not security boundary; artifact validated on load |
| T-23-02-02 | Repudiation | No audit trail for calibration capture | accept | calibrated_at_iso8601 field provides human-readable trace; not a security audit |

| T-23-02-SC | Tampering | npm/pip/cargo installs | accept | No new packages; uses stdlib json, pathlib, datetime only |
</threat_model>
